import math

print("program menghitung luas/keliling bidang datar\n")
print("1. persegi")
print("2. lingkaran")
print("3. segitiga\n")

pilihan = input("pilih: ")

if pilihan == "1":
    s = float(input("masukkan s: "))
    luas = s * s
    keliling = 4 * s
    print(f"luas: {luas}")
    print(f"keliling: {keliling}")

elif pilihan == "2":
    r = float(input("masukkan r: "))
    luas = math.pi * r * r
    keliling = 2 * math.pi * r
    print(f"luas: {luas}")
    print(f"keliling: {keliling}")

elif pilihan == "3":
    a = float(input("masukkan alas: "))
    t = float(input("masukkan tinggi: "))
    s1 = float(input("masukkan sisi 1: "))
    s2 = float(input("masukkan sisi 2: "))
    s3 = float(input("masukkan sisi 3: "))
    
    luas = 0.5 * a * t
    keliling = s1 + s2 + s3
    print(f"luas: {luas}")
    print(f"keliling: {keliling}")

else:
    print("pilihan nggak valid!")
